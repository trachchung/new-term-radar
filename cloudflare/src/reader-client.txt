async function api(path) {
  const session = access;
  const response = await fetch(path, {headers:{Authorization:"Bearer " + access},cache:"no-store"});
  if(!access || session !== access) throw Error("已退出查看，请重新输入访问密码。");
  if(response.status === 401) { logout(); throw Error("访问密码不正确，请重新输入。"); }
  if(!response.ok) throw Error(response.status === 404 ? "这份报告暂时不存在，请刷新归档列表。" : "读取失败，请稍后重试。");
  return response;
}
function inline(text, parent) {
  const pattern = /(\*\*([^*]+)\*\*|`([^`]+)`|\[([^\]]+)\]\((https?:\/\/[^\s)]+)\))/g;
  let end = 0, match;
  while((match=pattern.exec(text))) {
    parent.append(document.createTextNode(text.slice(end,match.index)));
    let element;
    if(match[2]) { element=document.createElement("strong"); element.textContent=match[2]; }
    else if(match[3]) { element=document.createElement("code"); element.textContent=match[3]; }
    else { element=document.createElement("a"); element.textContent=match[4]; element.href=match[5]; element.target="_blank"; element.rel="noopener noreferrer"; }
    parent.append(element); end=pattern.lastIndex;
  }
  parent.append(document.createTextNode(text.slice(end)));
}
function render(text) {
  const fragment=document.createDocumentFragment(), lines=text.replace(/\r/g,"").split("\n");
  const cells = line => line.trim().replace(/^\|/,"").replace(/\|$/,"").split("|").map(value=>value.trim());
  let i=0;
  while(i<lines.length) {
    const line=lines[i].trim();
    if(!line){i++;continue;}
    if(line.startsWith("|") && i+1<lines.length && /^\|[\s:|\-]+\|$/.test(lines[i+1].trim())) {
      const wrap=document.createElement("div");wrap.className="table-wrap";wrap.tabIndex=0;wrap.setAttribute("role","region");wrap.setAttribute("aria-label","报告数据表，可横向滚动");
      const table=document.createElement("table"), head=document.createElement("thead"), row=document.createElement("tr");
      cells(line).forEach(value=>{const cell=document.createElement("th");cell.scope="col";inline(tableLabels[value]||value,cell);row.append(cell);});head.append(row);table.append(head);
      const body=document.createElement("tbody");i+=2;
      while(i<lines.length&&lines[i].trim().startsWith("|")){const row=document.createElement("tr");cells(lines[i]).forEach(value=>{const cell=document.createElement("td");const key=value.replace(/^\*\*|\*\*$/g,"");inline(tableLabels[key]||value,cell);row.append(cell);});body.append(row);i++;}
      table.append(body);wrap.append(table);fragment.append(wrap);continue;
    }
    const heading=/^(#{1,6})\s+(.+)$/.exec(line);
    if(heading){const element=document.createElement("h"+Math.max(2,heading[1].length));inline(translations[heading[2]]||heading[2],element);fragment.append(element);i++;continue;}
    if(/^[-*]\s/.test(line)){const list=document.createElement("ul");while(i<lines.length&&/^[-*]\s/.test(lines[i].trim())){const item=document.createElement("li");inline(lines[i].trim().slice(2),item);list.append(item);i++;}fragment.append(list);continue;}
    const paragraph=document.createElement("p");inline(line,paragraph);fragment.append(paragraph);i++;
  }
  $("report").replaceChildren(fragment);
}
function drawDates() {
  const dates=[...new Map(entries.map(item=>[`${item.date}-${item.batch || 0}`,item])).values()];
  $("dates").replaceChildren();
  for(const group of dates){const date=group.date,batch=group.batch;const button=document.createElement("button");button.className="date-button";button.dataset.date=date;button.setAttribute("aria-current",selected?.date===date&&selected?.batch===batch?"date":"false");
    const label=document.createElement("span");label.textContent=batch?`${date}-${batch}`:date;
    const detail=document.createElement("small");detail.textContent=entries.filter(item=>item.date===date&&item.batch===batch).map(item=>item.kind==="daily"?"日报":"来源更新").join(" · ");button.append(label,detail);button.onclick=()=>selectReport(entries.find(item=>item.date===date&&item.batch===batch&&item.kind==="daily")||group);$("dates").append(button);}
  $("archive-status").textContent=dates.length?`${dates.length} 个报告批次`:"还没有报告。任务完成后会自动出现在这里。";
  $("more").hidden=!cursor;
}
function selectDate(date) {
  const item=entries.find(item=>item.date===date&&item.kind==="daily")||entries.find(item=>item.date===date);
  if(!item){requestNumber++;selected=null;markdown="";$("report").replaceChildren();$("report-tabs").replaceChildren();$("download").disabled=true;$("report-date").textContent=date;$("report-meta").textContent="";$("reading-status").className="reading-status";$("reading-status").textContent="这个日期还没有报告，请选择归档列表中的日期。";drawDates();return;}
  return selectReport(item);
}
async function selectReport(item) {
  selected=item;const number=++requestNumber;markdown="";$("download").disabled=true;$("report").replaceChildren();
  $("report-date").textContent=item.batch?`${item.date}-${item.batch}`:item.date;$("date-filter").value=item.date;
  $("report-meta").textContent=`${titles[item.kind]} · 更新时间 ${new Intl.DateTimeFormat("zh-CN",{timeZone:"Asia/Shanghai",hour:"2-digit",minute:"2-digit"}).format(new Date(item.updatedAt))}（北京时间）`;
  $("report-tabs").replaceChildren();
  for(const entry of entries.filter(entry=>entry.date===item.date&&entry.batch===item.batch)){const button=document.createElement("button");button.textContent=titles[entry.kind];button.setAttribute("aria-pressed",String(entry.kind===item.kind));button.onclick=()=>selectReport(entry);$("report-tabs").append(button);}
  drawDates();$("reading-status").className="reading-status loading";$("reading-status").textContent="正在读取报告…";
  try{const response=await api(`/api/report?date=${encodeURIComponent(item.date)}&kind=${item.kind}${item.batch?`&batch=${item.batch}`:""}`);const text=await response.text();if(number!==requestNumber)return;markdown=text;render(text);$("reading-status").textContent="";$("download").disabled=false;}
  catch(error){if(number===requestNumber)$("reading-status").textContent=error.message;}
  finally{if(number===requestNumber)$("reading-status").className="reading-status";}
}

  if(loading)return;loading=true;$("refresh").disabled=true;$("more").disabled=true;
  try{const response=await api("/api/reports"+(more&&cursor?"?cursor="+encodeURIComponent(cursor):""));const data=await response.json();entries=more?[...entries,...data.reports]:data.reports;cursor=data.cursor;drawDates();
    if(!more&&entries.length){const current=selected&&entries.find(item=>item.date===selected.date&&item.batch===selected.batch&&item.kind===selected.kind);await selectReport(current||entries[0]);}
    if(!access)throw Error("请重新输入访问密码。");
    if(!entries.length){$("report-date").textContent="等待第一期报告";$("report-meta").textContent="";$("report").replaceChildren();$("report-tabs").replaceChildren();$("download").disabled=true;}
  }finally{loading=false;$("refresh").disabled=false;$("more").disabled=false;}
}
